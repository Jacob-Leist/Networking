iptables script (with roll back)

# Save curent rules (before script) #
iptables-save > /etc/rules.bs

# Clear rules #
iptables -F
iptables -X

# Authorize already established connections
iptables -A INPUT -m state --state RELATED,ESTABLISHED -j ACCEPT
iptables -A OUTPUT -m state --state RELATED,ESTABLISHED -j ACCEPT

# Allow all used ports #
iptables -A INPUT -p tcp -m multiport --dports 20,21,25,53,80,88,110,135,139,143,389,443,5432,993,995 -j ACCEPT
iptables -A INPUT -p udp -m multiport --dports 53,80,88,123,135,139,389,443 -j ACCEPT
iptables -A INPUT -p tcp -m multiport --dports 445,464,465,597,636,3306,3268,3269,3389,5985,5986,8080,9997,587 -j ACCEPT
iptables -A INPUT -p udp -m multiport --dports 445,464,636,3268,3269,5985,5986,8080 -j ACCEPT

iptables -A FORWARD -p tcp -m multiport --dports 20,21,25,53,80,88,110,135,139,143,389,443 -j ACCEPT
iptables -A FORWARD -p tcp -m multiport --dports 445,464,465,597,636,3306,3268,3269,3389,5985,5986,8080,9997 -j ACCEPT
iptables -A FORWARD -p udp -m multiport --dports 53,80,88,123,135,139,389,443,445,464,636,3268,3269,5985,5986 -j ACCEPT

# allow remote connections #
iptables -A INPUT -p tcp -m multiport --dports 22,3389 -j ACCEPT
iptables -A INPUT -p udp -m multiport --dports 22,3389 -j ACCEPT
iptables -A FORWARD -p tcp -m multiport --dports 22,3389 -j ACCEPT
iptables -A FORWARD -p udp -m multiport --dports 22,3389 -j ACCEPT

# allow all #
iptables -A INPUT -j ACCEPT
iptables -A OUTPUT -j ACCEPT
iptables -A FORWARD -j ACCEPT

# Block everything by default
iptables -t filter -P INPUT DROP
iptables -t filter -P FORWARD DROP
iptables -t filter -P OUTPUT DROP

# Allow Ping #
iptables -t filter -A INPUT -p icmp -j ACCEPT
iptables -t filter -A OUTPUT -p icmp -j ACCEPT

# No port scaning #
iptables -N port-scanning
iptables -A port-scanning -p tcp --tcp-flags SYN,ACK,FIN,RST RST -m limit --limit 1/s --limit-burst 2 -j RETURN
iptables -A port-scanning -j DROP

# SSH attack protection #
iptables -A INPUT -p tcp --dport 22 -m conntrack --ctstate NEW -m limit --limit 1/min --limit-burst 5 -j ACCEPT
iptables -A OUTPUT -p tcp --dport 22 -m conntrack --ctstate NEW -m limit --limit 1/min --limit-burst 5 -j ACCEPT
iptables -A FORWARD -p tcp --dport 22 -m conntrack --ctstate NEW -m limit --limit 1/min --limit-burst 5 -j ACCEPT

# Save curent rules (after script) #
iptables-save > /etc/rules.as

iptables -L -v -n
